+"""Write the OpenAPI schema to a file without starting the server (AD-3, AD-19).
+
+Usage: `uv run python -m app.openapi_export <output-path>`
+"""
+
+import json
+import os
+import sys
+from pathlib import Path
+
+# Building the app validates settings; the schema needs no real secret.
+os.environ.setdefault("JWT_SECRET", "openapi-export")
+
+
+def export_openapi(output: Path) -> None:
+    from app.main import create_app
+
+    schema = create_app().openapi()
+    output.parent.mkdir(parents=True, exist_ok=True)
+    # Sorted keys, LF endings and a final newline keep the committed file stable.
+    text = json.dumps(schema, indent=2, sort_keys=True, ensure_ascii=False)
+    output.write_text(text + "\n", encoding="utf-8", newline="\n")
+
+
+def main(argv: list[str]) -> int:
+    if len(argv) != 2:
+        print("usage: python -m app.openapi_export <output-path>", file=sys.stderr)
+        return 2
+    export_openapi(Path(argv[1]))
+    return 0
+
+
+if __name__ == "__main__":
+    raise SystemExit(main(sys.argv))

+import type { ParseKeys } from 'i18next';
+import { useTranslation } from 'react-i18next';
+import { getErrorCode } from '@bite-and-board/data-access';
+
+/**
+ * Turns an API error into Hebrew copy (AD-28): `errors.<code>` from he.json,
+ * and the generic message for a missing or unknown code.
+ */
+export function useApiErrorMessage() {
+  const { t, i18n } = useTranslation();
+  return (error: unknown) => {
+    const code = getErrorCode(error);
+    // The code comes from the server, so the key is checked at runtime, not by the typed keys.
+    const key = code ? (`errors.${code}` as ParseKeys) : undefined;
+    return key && i18n.exists(key) ? t(key) : t('errors.unknown');
+  };
+}

+import { client } from './generated/client.gen';

+export const CSRF_HEADER = 'X-Requested-With';
+export const CSRF_VALUE = 'bite-and-board';
+
+let configured = false;
+
+/**
+ * Configures the one generated client (AD-19): same-origin cookies, and the
+ * CSRF header on every non-GET request. Safe to call more than once.
+ */
+export function configureClient() {
+  if (configured) return;
+  configured = true;
+  client.setConfig({ credentials: 'same-origin' });
+  client.interceptors.request.use((request) => {
+    if (request.method !== 'GET') {
+      request.headers.set(CSRF_HEADER, CSRF_VALUE);
+    }
+    return request;

+/** Every code the API publishes (AD-28). The UI renders `errors.<code>`. */
+export type { ErrorCode };
+
+/**
+ * Reads the `code` of a non-422 API error body `{detail, code, params?}`.
+ * Returns undefined for anything else (network failure, 422, unknown shape);
+ * a code newer than this client is returned as-is so the UI can fall back.
+ */
+export function getErrorCode(error: unknown): ErrorCode | undefined {
+  if (typeof error !== 'object' || error === null) return undefined;
+  const { code } = error as { code?: unknown };
+  return typeof code === 'string' ? (code as ErrorCode) : undefined;
